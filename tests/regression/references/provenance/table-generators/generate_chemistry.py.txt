"""Generate only card-physics masses and NIST energy/attenuation columns."""
import argparse
import hashlib
import json
from pathlib import Path
import re

ROOT=Path(__file__).resolve().parents[2]
SOURCES=['share/qualx/files/AtomProperties.xen','legacy/fortran/crystallography/nist_table.f90']
OUTPUT=ROOT/'src/core/crystallography/chemistry_data.inc'
META=ROOT/'docs/refactoring/phase-18/chemistry-resource-hashes.json'

# Logical source names remain the immutable resource provenance keys.
# Only their physical location changes after Native provider retirement.
def resource_path(name):
    if name.startswith('legacy/fortran/'):
        return ROOT/'archive/legacy/native-provider/fortran'/(name[len('legacy/fortran/'):]+'.txt')
    return ROOT/name


def data():
    rows=[s.split() for s in resource_path(SOURCES[0]).read_text().splitlines()[2:] if s.strip() and not s.startswith('#')]
    assert len(rows)%9==0
    masses=[dict(label=rows[i][0].capitalize(),z=int(rows[i][1]),mass=float(rows[i][2])) for i in range(0,len(rows),9)]
    text=resource_path(SOURCES[1]).read_text().split('   subroutine load_nist_table(zval)')[1]
    tables=[]
    for z,block in re.findall(r'case\((\d+)\)(.*?)(?=\n\s*case|end select)',text,re.S):
        arr=re.sub(r'!.*','',block.split('nist_tab(:) = [')[1].split(']')[0])
        v=[float(x) for x in re.findall(r'[-+]?\d*\.?\d+(?:[Ee][-+]?\d+)?',arr)]
        assert len(v)%4==0 and int(z)==len(tables)+1
        points=[(v[i],v[i+3]) for i in range(0,len(v),4)]
        assert all(a[0]<b[0] for a,b in zip(points,points[1:])) and all(x>0 and y>=0 for x,y in points)
        tables.append(points)
    assert len(masses)==299 and len(tables)==92
    assert [m['z'] for m in masses[:111]]==list(range(1,112))
    return masses,tables

def generate():
    masses,tables=data();fmt=lambda x:format(x,'.17g')
    out=['// Generated by tools/crystallography/generate_chemistry.py; do not edit.',
         '// XEN species index is shared with the immutable Phase 16 registry.',
         'constexpr ChemistryRecord chemistry_records[] = {']
    out += ['    {"%s", %d, %s},'%(m['label'],m['z'],fmt(m['mass'])) for m in masses]
    out += ['};','constexpr AttenuationPoint attenuation_points[] = {']
    for points in tables:out += ['    {%s, %s},'%(fmt(x),fmt(y)) for x,y in points]
    out += ['};','constexpr AttenuationTable attenuation_tables[] = {']
    offset=0
    for points in tables:
        out.append('    {%d, %d},'%(offset,len(points)));offset+=len(points)
    out += ['};',''];content='\n'.join(out)
    metadata=dict(sources={p:hashlib.sha256(resource_path(p).read_bytes()).hexdigest() for p in SOURCES},
        generator_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        output_sha256=hashlib.sha256(content.encode()).hexdigest(),species_masses=len(masses),
        attenuation_atomic_numbers=len(tables),attenuation_points=offset,
        units=dict(mass='dalton; numerically g/mol of atoms',energy='keV',mass_attenuation='cm^2/g'),
        resource_semantics='Original decimal resource values interpreted as binary64; no manual re-entry or modern-table substitution',
        nist_provenance='Legacy nist_table.f90, generated from NIST FFAST; original URL and generation notes retained in source',
        omitted=['f1/f2 (already in Phase 16)','bond/valence tables','UI descriptions','obsolete mass-absorption spline table','CIF data'],
        ranges=[dict(z=i+1,minimum_energy=p[0][0],maximum_energy=p[-1][0]) for i,p in enumerate(tables)])
    return content,json.dumps(metadata,indent=2)+'\n'

if __name__=='__main__':
    ap=argparse.ArgumentParser();ap.add_argument('--check',action='store_true');a=ap.parse_args()
    content,metadata=generate()
    if a.check:
        # Verify the sealed generator's historical identity separately from this
        # current reader. All resource values and other metadata bytes still
        # compare exactly with the original authority.
        original=ROOT/'archive/legacy/native-provider/tools/crystallography/generate_chemistry.py.txt'
        checked=json.loads(metadata)
        checked['generator_sha256']=hashlib.sha256(original.read_bytes()).hexdigest()
        metadata=json.dumps(checked,indent=2)+'\n'
    for path,text in zip((OUTPUT,META),(content,metadata)):
        if a.check:
            if path.read_bytes()!=text.encode():raise RuntimeError(str(path)+' differs; regenerate explicitly')
        else:path.write_bytes(text.encode())
    print('Chemistry data verified' if a.check else 'Chemistry data generated')
