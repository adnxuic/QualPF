def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--runner', type=Path, required=True)
    ap.add_argument('--output', type=Path, required=True)
    ap.add_argument('--only-prefix')
    args = ap.parse_args()
    runner = args.runner.resolve()
    out = args.output.resolve()
    out.mkdir(parents=True, exist_ok=False)
    sources = list(fixtures())
    for name in ('example1', 'CPD2', 'MIXT_5'):
        sources.append((name+'.dat', (ROOT/'share/qualx/examples'/f'{name}.dat').read_bytes()))
    sources.append(('inaccessible.xy', None))
    if args.only_prefix:
        sources = [(name, contents) for name, contents in sources if name.startswith(args.only_prefix)]
    cases = []
    for filename, content in sources:
        case = out/filename.replace('.', '-')
        case.mkdir()
        path = case/filename
        if content is not None:
            path.write_bytes(content if isinstance(content, bytes) else content.encode())
        before = {p.name:sha(p) for p in case.iterdir()}
        try:
            process = subprocess.run([str(runner), str(path)], cwd=case, capture_output=True, timeout=10)
            stdout, stderr, code = process.stdout, process.stderr, process.returncode
        except subprocess.TimeoutExpired as e:
            stdout, stderr, code = e.stdout or b'', e.stderr or b'', 'timeout'
        after = {p.name:sha(p) for p in case.iterdir()}
        (case/'stdout.txt').write_bytes(stdout)
        (case/'stderr.txt').write_bytes(stderr)
        status = 'abnormal'
        if b'PATTERN11' in stdout:
            data = stdout.split(b'PATTERN11', 1)[1].split()
            status = 'accepted' if data[0] == b'0' else 'rejected'
        item = dict(id=case.name, input=filename, returncode=code, native_status=status,
                    input_sha256=sha(path) if path.exists() else None,
                    stdout_sha256=sha(case/'stdout.txt'), stderr_sha256=sha(case/'stderr.txt'),
                    side_effects={k:v for k,v in after.items() if before.get(k)!=v})
        cases.append(item)
        print(case.name, status, code, flush=True)
    provenance = dict(source_commit=subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip(),
        runner_sha256=sha(runner), compiler='GNU Fortran 13.1.0; existing Qt MinGW product build; -O0',
        source_hashes={str(p.relative_to(ROOT)).replace('\\','/'):sha(p) for p in (ROOT/'legacy').rglob('*') if p.suffix.lower() in ('.f','.f90')},
        capture_sources={p.name:sha(p) for p in [Path(__file__),Path(__file__).with_suffix('.f90'),Path(__file__).with_name('pattern_capture_main.cpp')]})
    (out/'manifest.json').write_text(json.dumps(dict(schema_version=1,status='captured',provenance=provenance,cases=cases),indent=2)+'\n')


if __name__ == '__main__':
    main()
