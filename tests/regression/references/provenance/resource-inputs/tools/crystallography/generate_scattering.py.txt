"""Generate only active scattering data from retained, hash-pinned sources.

No chemistry metadata, obsolete fCu/fMo, or inactive electron fits are emitted.
Unsuffixed Fortran Lobato literals have binary32 literal semantics even though
stored in REAL(DP); expand those literal values exactly into binary64 constants.
Other resource decimals are parsed directly as binary64, not via Legacy Getnum.
"""
import argparse
import hashlib
import json
from pathlib import Path
import re
import struct

ROOT=Path(__file__).resolve().parents[2]
PATHS=['share/qualx/files/AtomProperties.xen',
       'legacy/fortran/crystallography/scatt_params.F90',
       'legacy/fortran/crystallography/nist_table.f90']
OUTPUT=ROOT/'src/core/crystallography/scattering_data.inc'
META=ROOT/'docs/refactoring/phase-16/table-resource-hashes.json'


def data():
    lines=(ROOT/PATHS[0]).read_text().splitlines()[2:]
    rows=[x.split() for x in lines if x.strip() and not x.startswith('#')]
    assert len(rows)%9==0
    species=[]
    for i in range(0,len(rows),9):
        label=rows[i][0].capitalize()
        charge=re.search(r'(\d+)([+-])$',label)
        charge=int(charge[1])*(1 if charge[2]=='+' else -1) if charge else 0
        v=list(map(float,rows[i+1][1:]+rows[i+2][1:]))
        assert len(v)==9
        species.append(dict(label=label,z=int(rows[i][1]),charge=charge,
                            a=v[:8:2],b=v[1:8:2],c=v[8],neutron=float(rows[i+7][1])))
    text=(ROOT/PATHS[1]).read_text()
    block=text.split('   subroutine get_lobato_params(')[1].split('   end subroutine get_lobato_params')[0]
    lobato=[]
    for z,arr in re.findall(r'case\((\d+)\)\s*fact\(:\) = \[\s*&?(.*?)\]',block,re.S):
        raw=re.findall(r'[-+]?\d+\.\d+[eE][-+]?\d+',arr)
        values=[struct.unpack('f',struct.pack('f',float(x)))[0] for x in raw]
        assert len(values)==10 and int(z)==len(lobato)+1
        lobato.append(values)
    assert len(lobato)==103
    text=(ROOT/PATHS[2]).read_text().split('   subroutine load_nist_table(zval)')[1]
    nist=[]
    for z,block in re.findall(r'case\((\d+)\)(.*?)(?=\n\s*case|end select)',text,re.S):
        frel=float(re.search(r'frel\s*=\s*([^\s]+)',block)[1])
        arr=re.sub(r'!.*','',block.split('nist_tab(:) = [')[1].split(']')[0])
        values=[float(x) for x in re.findall(r'[-+]?\d*\.?\d+(?:[Ee][-+]?\d+)?',arr)]
        assert len(values)%4==0 and int(z)==len(nist)+1
        # Mass attenuation is not part of the structure-factor domain.
        points=[values[j:j+3] for j in range(0,len(values),4)]
        assert all(a[0]<b[0] for a,b in zip(points,points[1:]))
        nist.append(dict(z=int(z),frel=frel,points=points))
    assert len(nist)==92 and len(species)==299
    return species,lobato,nist


def generate():
    species,lobato,nist=data()
    num=lambda v:format(v,'.17g')
    arr=lambda vs:'{'+', '.join(map(num,vs))+'}'
    out=['// Generated by tools/crystallography/generate_scattering.py; do not edit.',
         '// Provenance and units: docs/refactoring/phase-16/table-resource-hashes.json',
         'constexpr SpeciesRecord species_records[] = {']
    for s in species:
        out.append('    {"%s", %d, %d, %s, %s, %s, %s},' %
                   (s['label'],s['z'],s['charge'],arr(s['a']),arr(s['b']),num(s['c']),num(s['neutron'])))
    out += ['};','constexpr std::array<double, 10> lobato_records[] = {']
    out += ['    '+arr(row)+',' for row in lobato]
    out += ['};','constexpr NistPoint nist_points[] = {']
    for table in nist:
        out += ['    '+arr(row)+',' for row in table['points']]
    out += ['};','constexpr NistRecord nist_records[] = {']
    offset=0
    for table in nist:
        out.append('    {%d, %d, %s},' % (offset,len(table['points']),num(table['frel'])))
        offset+=len(table['points'])
    out += ['};','']
    content='\n'.join(out)
    metadata=dict(sources={p:hashlib.sha256((ROOT/p).read_bytes()).hexdigest() for p in PATHS},
        generator_sha256=hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        output_sha256=hashlib.sha256(content.encode()).hexdigest(),species=len(species),
        xray_entries=sum(any(s['a']) or s['c']!=0 for s in species),
        neutron_entries=sum(s['neutron']!=0 for s in species),
        electron_entries=sum(s['z']<=103 for s in species),electron_atomic_numbers=103,
        anomalous_atomic_numbers=92,nist_points=offset,
        units=dict(xray='electrons',neutron='10^-12 cm',electron='retained Lobato amplitude at s² in Angstrom^-2'),
        constant_semantics='XEN and NIST decimals binary64; Lobato unsuffixed REAL literals exactly widened to binary64; all runtime arithmetic binary64',
        unsupported='All-zero X-ray fits and zero neutron placeholders are unavailable; electron Z>103 and anomalous Z>92 unavailable',
        omitted=['chemistry UI metadata','bond tables','mass attenuation','obsolete fMo/fCu','inactive electron fits'])
    return content,json.dumps(metadata,indent=2)+'\n'


if __name__=='__main__':
    ap=argparse.ArgumentParser(); ap.add_argument('--check',action='store_true'); args=ap.parse_args()
    content,meta=generate()
    for path,text in ((OUTPUT,content),(META,meta)):
        if args.check:
            assert path.read_bytes()==text.encode(),str(path)+' differs; regenerate explicitly'
        else: path.write_bytes(text.encode())
    print('Scattering data verified' if args.check else 'Scattering data generated')
