cmake_minimum_required(VERSION 3.20)
project(QualpfApplicationContracts LANGUAGES CXX)
enable_testing()
set(QUALPF_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(QUALPF_BUILD_PRODUCT_TESTS OFF)
add_subdirectory("${QUALPF_ROOT}/src/core" core)
# The same complete Application target used by the product. No Qt discovery,
# compile definition, include directory or linked binary enters this harness.
add_subdirectory("${QUALPF_ROOT}/src/application" application)
add_executable(application_boundary application_boundary.cpp)
target_link_libraries(application_boundary PRIVATE qualpf_application)
add_test(NAME application.qtfree COMMAND application_boundary)
add_executable(geometric_fwhm_core ../unit/core/geometric_fwhm_test.cpp)
target_link_libraries(geometric_fwhm_core PRIVATE qualpf_core)
add_test(NAME geometric_fwhm.core COMMAND geometric_fwhm_core)
add_executable(geometric_fwhm_application ../unit/application/geometric_fwhm_test.cpp)
target_link_libraries(geometric_fwhm_application PRIVATE qualpf_session)
add_test(NAME geometric_fwhm.application COMMAND geometric_fwhm_application)
add_executable(core_algorithm_modules ../unit/core/algorithm_modules_test.cpp)
target_include_directories(core_algorithm_modules PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../support")
target_link_libraries(core_algorithm_modules PRIVATE qualpf_core)
add_test(NAME algorithms.core_modules COMMAND core_algorithm_modules)
add_executable(configured_fwhm ../unit/application/configured_fwhm_test.cpp)
target_link_libraries(configured_fwhm PRIVATE qualpf_session)
add_test(NAME fwhm.configured COMMAND configured_fwhm)
add_executable(initial_peak_search ../unit/application/initial_peak_search_test.cpp)
target_link_libraries(initial_peak_search PRIVATE qualpf_session)
add_test(NAME peaks.initial COMMAND initial_peak_search)
add_executable(pfsm_core ../unit/core/pfsm_test.cpp ../unit/core/pfsm_candidate_recall_test.cpp)
target_include_directories(pfsm_core PRIVATE "${QUALPF_ROOT}/src/core/profile")
target_link_libraries(pfsm_core PRIVATE qualpf_core)
if(MSVC)
  target_compile_options(pfsm_core PRIVATE /fp:strict)
else()
  target_compile_options(pfsm_core PRIVATE -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME pfsm.core COMMAND pfsm_core)
add_executable(pfsm_probe ../unit/core/pfsm_probe.cpp)
add_executable(pfsm_candidate_recall_probe ../unit/core/pfsm_candidate_recall_probe.cpp)
target_include_directories(pfsm_candidate_recall_probe PRIVATE "${QUALPF_ROOT}/src/core/profile")
target_link_libraries(pfsm_candidate_recall_probe PRIVATE qualpf_core)
target_link_libraries(pfsm_probe PRIVATE qualpf_core)
add_executable(pfsm_application ../unit/application/pfsm_test.cpp)
find_package(OpenMP REQUIRED COMPONENTS CXX)
target_link_libraries(pfsm_application PRIVATE qualpf_application OpenMP::OpenMP_CXX)
add_test(NAME pfsm.application COMMAND pfsm_application)
add_executable(profile_rir ../unit/core/profile_rir_test.cpp)
target_link_libraries(profile_rir PRIVATE qualpf_core)
add_test(NAME quantification.core COMMAND profile_rir)
add_executable(pfsm_quantification ../unit/application/pfsm_quantification_test.cpp)
target_link_libraries(pfsm_quantification PRIVATE qualpf_application)
add_test(NAME quantification.application COMMAND pfsm_quantification)

add_executable(quantification_export ../unit/application/quantification_export_test.cpp)
target_link_libraries(quantification_export PRIVATE qualpf_application)
add_test(NAME quantification.export COMMAND quantification_export)
