cmake_minimum_required(VERSION 3.20)
project(QualpfProfileContracts LANGUAGES CXX)
enable_testing()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    add_subdirectory(../../src/core core)
endif()
if(NOT TARGET qualpf::profile)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../src/core/profile ${CMAKE_CURRENT_BINARY_DIR}/profile)
endif()
add_executable(structure_profile_replay structure_replay.cpp)
target_link_libraries(structure_profile_replay PRIVATE qualpf::profile_structure)
add_executable(profile_baseline baseline.cpp)
target_link_libraries(profile_baseline PRIVATE qualpf::profile)
if(WIN32)
  target_link_libraries(profile_baseline PRIVATE psapi)
endif()
add_executable(profile_replay replay.cpp)
target_link_libraries(profile_replay PRIVATE qualpf::profile)
add_executable(profile_properties properties.cpp)
target_link_libraries(profile_properties PRIVATE qualpf::profile)
if(MSVC)
  target_compile_options(structure_profile_replay PRIVATE /fp:strict)
  target_compile_options(profile_replay PRIVATE /fp:strict)
  target_compile_options(profile_properties PRIVATE /fp:strict)
else()
  target_compile_options(structure_profile_replay PRIVATE -fno-fast-math -ffp-contract=off)
  target_compile_options(profile_replay PRIVATE -fno-fast-math -ffp-contract=off)
  target_compile_options(profile_properties PRIVATE -fno-fast-math -ffp-contract=off)
endif()
add_test(NAME qualpf.profile.properties COMMAND profile_properties)
set_tests_properties(qualpf.profile.properties PROPERTIES LABELS normal)
