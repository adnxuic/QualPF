# Deliberately separate from discovery (cifreader.cpp) and crystallographic semantics.
add_library(qualpf_cif_syntax STATIC cif_document.cpp cif_tokenizer.cpp cif_parser.cpp)
add_library(qualpf::cif_syntax ALIAS qualpf_cif_syntax)
target_compile_features(qualpf_cif_syntax PUBLIC cxx_std_20)
target_include_directories(qualpf_cif_syntax PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(qualpf_cif_syntax PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(MSVC)
    target_compile_options(qualpf_cif_syntax PRIVATE /W4 /fp:strict)
else()
    target_compile_options(qualpf_cif_syntax PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()

if(TARGET qualpf_core)
    add_library(qualpf_cif_import STATIC cif_symmetry.cpp cif_import.cpp cif_provenance.cpp cif_site_symmetry.cpp)
    add_library(qualpf::cif_import ALIAS qualpf_cif_import)
    target_link_libraries(qualpf_cif_import PUBLIC qualpf_cif_syntax qualpf_core)
    target_compile_features(qualpf_cif_import PUBLIC cxx_std_20)
    set_target_properties(qualpf_cif_import PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        target_compile_options(qualpf_cif_import PRIVATE /W4 /fp:strict)
    else()
        target_compile_options(qualpf_cif_import PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    endif()
endif()
