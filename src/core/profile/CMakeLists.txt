target_include_directories(qualpf_profile PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
target_compile_features(qualpf_profile PUBLIC cxx_std_20)
set_target_properties(qualpf_profile PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)

endif()
